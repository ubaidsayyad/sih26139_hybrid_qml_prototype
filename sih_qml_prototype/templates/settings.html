● Quick Start Integration

Calling the Clinical Prediction API

# Python Example using requests
import requests

api_key = "{{ current_user.api_key }}"
headers = {
    "X-API-Key": api_key,
    "Content-Type": "application/json"
}
payload = {
    "features": [0.2845, -0.1492, 0.0821, -0.0195],
    "dataset": "Breast Cancer Wisconsin (Diagnostic)"
}

resp = requests.post("http://localhost:5000/api/v1/predict", json=payload, headers=headers)
print(resp.json())
